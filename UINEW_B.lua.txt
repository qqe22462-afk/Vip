local function uiLockRead()
local data = _G[UI_LOCK_KEY]
if type(data) ~= "table" then return nil end
return data
end
local function uiLockWrite()
_G[UI_LOCK_KEY] = {
ownerId   = UI_SESSION_ID,
lastBeat  = os.clock(),
startedAt = os.time(),
}
end
local function uiLockIsStale(data)
if type(data) ~= "table" then return true end
if type(data.lastBeat) ~= "number" then return true end
return (os.clock() - data.lastBeat) > UI_LOCK_TIMEOUT
end
local function uiLockAcquire()
local cur = uiLockRead()
if cur == nil or uiLockIsStale(cur) or cur.ownerId == UI_SESSION_ID then
uiLock.isOwner  = true
uiLock.ownerId  = UI_SESSION_ID
uiLock.lastBeat = os.clock()
uiLockWrite()
return true
end
uiLock.isOwner  = false
uiLock.ownerId  = cur.ownerId
uiLock.lastBeat = cur.lastBeat or 0
return false
end
local function uiLockRelease()
local cur = uiLockRead()
if type(cur) == "table" and cur.ownerId == UI_SESSION_ID then
_G[UI_LOCK_KEY] = nil
end
uiLock.isOwner = false
end

local function ResolveAssetUrl(cid)
if not RunService:IsStudio() then return nil end
local url
pcall(function()
local objs = game:GetObjects("rbxassetid://" .. cid)
if objs and #objs > 0 then
local o = objs[1]
if o:IsA("Decal") or o:IsA("Texture") then url = o.Texture
elseif o:IsA("ImageLabel") or o:IsA("ImageButton") then url = o.Image
else
local d = o:FindFirstChildWhichIsA("Decal", true)
or o:FindFirstChildWhichIsA("Texture", true)
if d then url = d.Texture end
end
pcall(function() o:Destroy() end)
end
end)
return url
end

end

local function safeColor3(r, g, b, fallback)
r, g, b = tonumber(r), tonumber(g), tonumber(b)
if not (r and g and b) then return fallback end
return Color3.fromRGB(
math.clamp(math.floor(r + 0.5), 0, 255),
math.clamp(math.floor(g + 0.5), 0, 255),
math.clamp(math.floor(b + 0.5), 0, 255)
)
end
local function colorToArray(c)
return {
math.floor(c.R * 255 + 0.5),
math.floor(c.G * 255 + 0.5),
math.floor(c.B * 255 + 0.5),
}
end
local function colorToHex(c)
local a = colorToArray(c)
return string.format("#%02X%02X%02X", a[1], a[2], a[3])
end
local function hexToColor3(hex, fallback)
if type(hex) ~= "string" then return fallback end
hex = hex:gsub("%s+", "")
if hex:sub(1, 1) == "#" then hex = hex:sub(2) end
if #hex ~= 6 then return fallback end
local r = tonumber(hex:sub(1, 2), 16)
local g = tonumber(hex:sub(3, 4), 16)
local b = tonumber(hex:sub(5, 6), 16)
if not (r and g and b) then return fallback end
return Color3.fromRGB(r, g, b)
end

end

local function colorsEqual(a, b)
return typeof(a) == "Color3" and typeof(b) == "Color3"
and math.abs(a.R - b.R) < 0.001
and math.abs(a.G - b.G) < 0.001
and math.abs(a.B - b.B) < 0.001
end
local function paletteRoleColor(role)
if role == "Background" then return Theme.Background
elseif role == "Surface" then return Theme.Surface
elseif role == "SurfaceAlt" then return Theme.SurfaceAlt
elseif role == "SurfaceHover" then return Theme.SurfaceHover
elseif role == "Accent" then return Theme.Accent
elseif role == "AccentLight" then return Theme.AccentLight
elseif role == "AccentSoft" then return Theme.AccentSoft
elseif role == "Text" then return Theme.Text
elseif role == "TextMuted" then return Theme.TextMuted
elseif role == "Border" then return Theme.Border
elseif role == "Success" then return Theme.Success
elseif role == "Danger" then return Theme.Danger
elseif role == "Warning" then return Theme.Warning
elseif role == "SwitchOff" then return Theme.SwitchOff
elseif role == "White" then return Theme.White
elseif role == "IconDim" then return Theme.IconDim
elseif role == "ToggleState" then return Theme.Accent
elseif role == "TabButton" then return Theme.Accent
elseif role == "SettingsButton" then return Theme.AccentSoft
end
return nil
end

c = selected and Theme.Accent or Theme.SurfaceAlt

if c and prop then
pcall(function() obj[prop] = c end)

end

local function collectSettings()

local function applyPosition(gui, data)
if not gui or type(data) ~= "table" then return end
local xs = tonumber(data.XScale) or gui.Position.X.Scale
local xo = tonumber(data.XOffset) or gui.Position.X.Offset
local ys = tonumber(data.YScale) or gui.Position.Y.Scale
local yo = tonumber(data.YOffset) or gui.Position.Y.Offset
gui.Position = UDim2.new(xs, xo, ys, yo)
end
local function applySettingsData(data)
if type(data) ~= "table" then return false end
UserScale = 1.0
UserTextScale = math.clamp(tonumber(data.TextScale) or DefaultConfig.TextScale, 1.00, 3.00)
local fw = tostring(data.FontWeight or DefaultConfig.FontWeight)
if not FONT_WEIGHTS[fw] then fw = DefaultConfig.FontWeight end
UserFontWeight = fw
if type(data.Theme) == "table" then
for key, fallback in pairs(DefaultTheme) do
if typeof(fallback) == "Color3" then
local v = data.Theme[key]
if type(v) == "table" then
Theme[key] = safeColor3(v[1], v[2], v[3], fallback)
end
end
end
end
if tonumber(data.Version) and tonumber(data.Version) >= 4 then
if data.MainPosition and mainFrame then
local xs = tonumber(data.MainPosition.XScale) or 0.5
local ys = tonumber(data.MainPosition.YScale) or 0.5
if xs > 0.01 and xs < 0.99 and ys > 0.01 and ys < 0.99 then
applyPosition(mainFrame, data.MainPosition)
else
mainFrame.AnchorPoint = Vector2.new(0.5, 0.5)
mainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
end
end
if data.PillPosition and TogglePill then
local xs = tonumber(data.PillPosition.XScale) or 1
local ys = tonumber(data.PillPosition.YScale) or 0
if xs > 0.01 and xs < 0.99 and ys > 0.01 and ys < 0.99 then
applyPosition(TogglePill, data.PillPosition)
end
end
else
if mainFrame then
mainFrame.AnchorPoint = Vector2.new(0.5, 0.5)
mainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
end
end
applySettingsVisuals()
return true
end
local function saveSettings()
local data = collectSettings()
MemorySettings = data
LastLoadedFromFile = false
if not hasFileStorage() then return false, "ระบบนี้ไม่มี writefile/isfile; เก็บไว้ในหน่วยความจำของเซสชันนี้" end
local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
if not ok then return false, "แปลงข้อมูลการตั้งค่าเป็น JSON ไม่สำเร็จ" end
local okWrite, err = pcall(function() writefile(SETTINGS_FILE, encoded) end)
if not okWrite then return false, "เขียนไฟล์ไม่สำเร็จ: " .. tostring(err) end
return true, SETTINGS_FILE
end
local function loadSettings()
if not hasFileStorage() or not isfile(SETTINGS_FILE) then
if MemorySettings then
applySettingsData(MemorySettings)
return true, "หน่วยความจำ"
end
return false, "ยังไม่พบไฟล์การตั้งค่า"
end
local okRead, raw = pcall(function() return readfile(SETTINGS_FILE) end)
if not okRead or type(raw) ~= "string" or raw == "" then
return false, "อ่านไฟล์การตั้งค่าไม่สำเร็จ"
end
local okDecode, data = pcall(function() return HttpService:JSONDecode(raw) end)
if not okDecode or type(data) ~= "table" then
return false, "ไฟล์การตั้งค่ามีรูปแบบไม่ถูกต้อง"
end
if not applySettingsData(data) then return false, "ใช้ข้อมูลการตั้งค่าไม่สำเร็จ" end
MemorySettings = data
LastLoadedFromFile = true
return true, SETTINGS_FILE
end

_G.RUNLUA_IS_DRAGGING = _G.RUNLUA_IS_DRAGGING or false
local function MakeDraggable(frame, opts)
opts = opts or {}
local touchHandle = opts.TouchHandle or opts.Handle or frame
local mouseHandle = opts.MouseHandle or opts.Handle or frame
local drag = {
activeInput   = nil,
dragMode      = nil,
dragStart     = nil,
startPos      = nil,
startAbsPos   = Vector2.new(0, 0),
movedDistance = 0,
}
local function resetDrag()
drag.activeInput   = nil
drag.dragMode      = nil
drag.dragStart     = nil
drag.startPos      = nil
drag.startAbsPos   = Vector2.new(0, 0)
drag.movedDistance = 0
_G.RUNLUA_IS_DRAGGING = false
end
local function isPointIn(h, pos)
if not h or not h.Parent then return false end
local ap = h.AbsolutePosition
local as = h.AbsoluteSize
return pos.X >= ap.X and pos.X <= (ap.X + as.X)
and pos.Y >= ap.Y and pos.Y <= (ap.Y + as.Y)
end
local function isPointOverButton(h, pos)
if not h then return false end
for _, child in ipairs(h:GetDescendants()) do
if child:IsA("GuiButton") and child.Visible and child.Active ~= false then
local ap = child.AbsolutePosition
local as = child.AbsoluteSize
if pos.X >= ap.X and pos.X <= ap.X + as.X
and pos.Y >= ap.Y and pos.Y <= ap.Y + as.Y then
return true
end
end
end
return false
end
track(UserInputService.InputBegan:Connect(function(input, gp)
if not frame or not frame.Parent then return end
if not uiLock.isOwner then return end
if drag.activeInput then return end
local pos = Vector2.new(input.Position.X, input.Position.Y)
if IS_MOBILE then
if gp then return end
if input.UserInputType ~= Enum.UserInputType.Touch then return end
if not isPointIn(touchHandle, pos) then return end
if isPointOverButton(touchHandle, pos) then return end
drag.activeInput   = input
drag.dragMode      = "touch"
drag.dragStart     = input.Position
drag.startPos      = frame.Position
drag.startAbsPos   = pos
drag.movedDistance = 0
_G.RUNLUA_IS_DRAGGING = true
else
if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
if not isPointIn(mouseHandle, pos) then return end
if isPointOverButton(mouseHandle, pos) then return end
drag.activeInput   = input
drag.dragMode      = "mouse"
drag.dragStart     = input.Position
drag.startPos      = frame.Position
drag.startAbsPos   = pos
drag.movedDistance = 0
_G.RUNLUA_IS_DRAGGING = true
end
end))
track(UserInputService.InputChanged:Connect(function(input, gp)
if not drag.activeInput or not drag.dragMode then return end
local it = input.UserInputType
local valid
if drag.dragMode == "mouse" then
valid = (it == Enum.UserInputType.MouseMovement)
elseif drag.dragMode == "touch" then
if gp then return end
valid = (input == drag.activeInput and it == Enum.UserInputType.Touch)
end
if not valid then return end
local currentPos = Vector2.new(input.Position.X, input.Position.Y)
local deltaMove  = (currentPos - drag.startAbsPos).Magnitude
if deltaMove > drag.movedDistance then
drag.movedDistance = deltaMove
end
local vp    = getViewport()
local inset = getSafeInset()
local startX = drag.startPos.X.Scale * vp.X + drag.startPos.X.Offset
local startY = drag.startPos.Y.Scale * vp.Y + drag.startPos.Y.Offset
local delta = input.Position - drag.dragStart
local newX, newY = startX + delta.X, startY + delta.Y
local halfW = frame.AbsoluteSize.X / 2
local halfH = frame.AbsoluteSize.Y / 2
newX = math.clamp(newX, halfW, vp.X - halfW)
newY = math.clamp(newY, inset.Y + halfH, vp.Y - halfH)
frame.Position = UDim2.new(0, newX, 0, newY)
end))
track(UserInputService.InputEnded:Connect(function(input)
if not drag.activeInput or not drag.dragMode then return end
local it = input.UserInputType
local ends
if drag.dragMode == "mouse" then
ends = (it == Enum.UserInputType.MouseButton1)
elseif drag.dragMode == "touch" then
ends = (input == drag.activeInput and it == Enum.UserInputType.Touch)
end
if not ends then return end
resetDrag()
end))
track(screenGui.Destroying:Connect(resetDrag))
end

local function isPointInResizeHandle(pos)
if not resizeHandle or not resizeHandle.Parent then return false end
local ap = resizeHandle.AbsolutePosition
local as = resizeHandle.AbsoluteSize
return pos.X >= ap.X and pos.X <= ap.X + as.X
and pos.Y >= ap.Y and pos.Y <= ap.Y + as.Y
end
track(resizeHandle.InputBegan:Connect(function(input)
if not uiLock.isOwner then return end
if resizeActive then return end
if _G.RUNLUA_IS_DRAGGING then return end
local it = input.UserInputType
if it ~= Enum.UserInputType.Touch
and it ~= Enum.UserInputType.MouseButton1 then
return
end
local pos = Vector2.new(input.Position.X, input.Position.Y)
if not isPointInResizeHandle(pos) then
return
end
resizeActive    = true
resizeInputObj  = input
resizeMode      = (it == Enum.UserInputType.Touch) and "touch" or "mouse"
resizeScale     = math.max(uiScale.Scale, 0.01)
resizeStartPos  = pos
resizeStartSize = Vector2.new(
mainFrame.AbsoluteSize.X / resizeScale,
mainFrame.AbsoluteSize.Y / resizeScale
)
end))

if resizeMode == "mouse" and it ~= Enum.UserInputType.MouseMovement then return end

local minW, minH = Config.WindowMinSize.X, Config.WindowMinSize.Y

local maxH = math.max(minH, (vp.Y - 24) / resizeScale)

local h = math.clamp(resizeStartSize.Y + delta.Y / resizeScale, minH, maxH)

local pillToggleWindow
pill = {
active        = false,
input         = nil,
mode          = nil,
dragStartPos  = nil,
pillStartPos  = nil,
movedDistance = 0,
lastToggle    = 0,
lastClick     = 0,

local function resetPillDrag()
pill.active        = false
pill.input         = nil
pill.mode          = nil
pill.dragStartPos  = nil
pill.pillStartPos  = nil
pill.movedDistance = 0
end
local function isPointInPill(pos)
local ap = TogglePill.AbsolutePosition
local as = TogglePill.AbsoluteSize
return pos.X >= ap.X and pos.X <= (ap.X + as.X)
and pos.Y >= ap.Y and pos.Y <= (ap.Y + as.Y)
end
pillToggleWindow = function()
if not uiLock.isOwner then
Notify({
Title = "UI ถูกล็อก",
Message = "มีสคริปต์อื่นถือสิทธิ์อยู่ (UISC=0)",
Color = Theme.Warning,
Icon = Icons.Action,
})
return
end
local now = os.clock()
if now - pill.lastToggle < Config.ToggleDebounce then return end
pill.lastToggle = now
if mainFrame.Visible then hideWindow() else showWindow() end
end
track(UserInputService.InputBegan:Connect(function(input)
if not TogglePill.Visible then return end
if pill.active then return end
local it = input.UserInputType
if it ~= Enum.UserInputType.Touch
and it ~= Enum.UserInputType.MouseButton1 then
return
end
local pos = Vector2.new(input.Position.X, input.Position.Y)
if not isPointInPill(pos) then return end
pill.active        = true
pill.input         = input
pill.mode          = (it == Enum.UserInputType.Touch) and "touch" or "mouse"
pill.dragStartPos  = pos
pill.pillStartPos  = TogglePill.Position
pill.movedDistance = 0
TweenService:Create(TogglePill, TweenInfo.new(0.1), {
Size = UDim2.fromOffset(pillSizeMin, pillSizeMin)
}):Play()
end))
track(UserInputService.InputChanged:Connect(function(input)
if not pill.active then return end
if input ~= pill.input then return end
local it = input.UserInputType
local valid
if pill.mode == "touch" then
valid = (it == Enum.UserInputType.Touch)
else
valid = (it == Enum.UserInputType.MouseMovement)
end
if not valid then return end
local currentPos = Vector2.new(input.Position.X, input.Position.Y)
local deltaMove  = (currentPos - pill.dragStartPos).Magnitude
if deltaMove > pill.movedDistance then
pill.movedDistance = deltaMove
end
local vp    = getViewport()
local inset = getSafeInset()
local startX = pill.pillStartPos.X.Scale * vp.X + pill.pillStartPos.X.Offset
local startY = pill.pillStartPos.Y.Scale * vp.Y + pill.pillStartPos.Y.Offset
local delta = Vector2.new(
input.Position.X - pill.dragStartPos.X,
input.Position.Y - pill.dragStartPos.Y
)
local newX = startX + delta.X
local newY = startY + delta.Y
local half = TogglePill.AbsoluteSize.X / 2
newX = math.clamp(newX, half, vp.X - half)
newY = math.clamp(newY, inset.Y + half, vp.Y - half)
TogglePill.Position = UDim2.new(0, newX, 0, newY)
end))
track(UserInputService.InputEnded:Connect(function(input)
if not pill.active then return end
if input ~= pill.input then return end
local it = input.UserInputType
local ends
if pill.mode == "touch" then
ends = (it == Enum.UserInputType.Touch)
else
ends = (it == Enum.UserInputType.MouseButton1)
end
if not ends then return end
local dragged = pill.movedDistance > Config.DragThreshold
TweenService:Create(TogglePill, TweenInfo.new(0.15, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
Size = UDim2.fromOffset(pillSize, pillSize)
}):Play()
local lastClick = pill.lastClick
resetPillDrag()
pill.lastClick = lastClick
if not dragged then
local now = os.clock()
if now - pill.lastClick >= Config.ToggleDebounce then
pill.lastClick = now
pcall(function() if pillToggleWindow then pillToggleWindow() end end)
end
end
end))
